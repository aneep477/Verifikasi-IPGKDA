// ============================================================ // KONFIGURASI GLOBAL // ============================================================ const SPREADSHEET_ID = '1N8fz2FSP6ypJcK0BN3BacSThdcZ7ckROVvVvoinj2sw'; const DRIVE_FOLDER_ID = '1A7NiVQVSfUlRI0vVue2NOuNYN8EqHypR'; // ============================================================ // DOGET - LAMAN UTAMA // ============================================================ function doGet() { return HtmlService.createTemplateFromFile('index') .evaluate() .addMetaTag('viewport', 'width=device-width, initial-scale=1') .setTitle('Sistem Verifikasi Pentaksiran Berterusan') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } function include(filename) { return HtmlService.createHtmlOutputFromFile(filename).getContent(); } // ============================================================ // MUAT NAIK FAIL KE DRIVE (DENGAN PENGESAHAN) // ============================================================ function uploadFileToDrive(base64Data, fileName, mimeType) { try { if (!base64Data || typeof base64Data !== 'string') { throw new Error('Data fail tidak sah (bukan string).'); } const folder = DriveApp.getFolderById(DRIVE_FOLDER_ID); const base64 = base64Data.split(',').pop(); const blob = Utilities.newBlob(Utilities.base64Decode(base64), mimeType, fileName); const file = folder.createFile(blob); return { success: true, url: file.getUrl(), name: file.getName() }; } catch (error) { console.error('Upload error:', error); return { success: false, error: error.toString() }; } } // ============================================================ // SIMPAN DATA BORANG (DENGAN PENGENDALIAN ARRAY -> STRING) // ============================================================ function saveFormData(formData, filesData, formType) { try { // Tukar sebarang nilai array dalam formData kepada string (pencegahan ralat "key must be string") const cleanedData = {}; for (let key in formData) { let value = formData[key]; if (Array.isArray(value)) { value = value.join(', '); // contoh: jika multiple select, gabungkan } cleanedData[key] = (value !== undefined && value !== null) ? value : ''; } const sheetName = (formType === 'perancangan') ? 'Perancangan' : 'SelepasModerasi'; let sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName(sheetName); if (!sheet) { createSheetIfNotExists(sheetName, formType); sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName(sheetName); } // Muat naik fail lampiran const uploadedUrls = []; if (filesData && Array.isArray(filesData) && filesData.length) { for (const file of filesData) { const result = uploadFileToDrive(file.data, file.name, file.mimeType); if (result.success) uploadedUrls.push(result.url); } } // Baca header dari sheet const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; const rowData = []; for (let i = 0; i < headers.length; i++) { const header = headers[i]; if (header === 'timestamp') { rowData.push(new Date()); } else if (header === 'lampiran_urls') { rowData.push(uploadedUrls.join(', ')); } else if (header === 'tarikh_taklimat_pb' && formType === 'selepasmoderasi') { rowData.push(cleanedData['tarikh_taklimat_pb'] || ''); } else { // Pastikan nilai adalah string (jika wujud) rowData.push(cleanedData[header] !== undefined ? cleanedData[header].toString() : ''); } } sheet.appendRow(rowData); return { success: true, message: 'Data berjaya disimpan!' }; } catch (error) { console.error('saveFormData error:', error); return { success: false, error: error.toString() }; } } // ============================================================ // CIPTA SHEET DENGAN HEADER LENGKAP (jika belum wujud) // ============================================================ function createSheetIfNotExists(sheetName, formType) { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); let sheet = ss.getSheetByName(sheetName); if (sheet) return; sheet = ss.insertSheet(sheetName); let headers = []; if (formType === 'perancangan') { headers = ['timestamp', 'ipg_kampus', 'kod_kursus', 'nama_kursus', 'program', 'semester_tahun', 'tarikh_verifikasi', 'ulasan_keseluruhan', 'cadangan_penambahbaikan', 'lampiran_urls']; for (let i = 1; i <= 21; i++) { headers.push(`item${i}`); headers.push(`item${i}_catatan`); } } else { headers = ['timestamp', 'ipg_kampus', 'kod_kursus', 'nama_kursus', 'program', 'semester_tahun', 'tarikh_verifikasi', 'tarikh_taklimat_pb', 'ulasan_keseluruhan', 'cadangan_penambahbaikan', 'lampiran_urls']; for (let i = 1; i <= 12; i++) { headers.push(`item${i}`); headers.push(`item${i}_catatan`); } } sheet.getRange(1, 1, 1, headers.length).setValues([headers]); sheet.setFrozenRows(1); } // Fungsi untuk setup awal (panggil sekali) function setupSheets() { createSheetIfNotExists('Perancangan', 'perancangan'); createSheetIfNotExists('SelepasModerasi', 'selepasmoderasi'); }
